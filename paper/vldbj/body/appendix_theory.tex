\section{Proofs for Section~\ref{sec:theory}}
\label{app:proofs}

Throughout, $\tilde X\in\mathbb R^{K\times P}$ is the folded normalized window with phase tokens $\tilde x_p$ and cycle means $\bar r$, $\one^\top\tilde x=0$, $\|\tilde x\|^2\le L$, $\one^\top\bar r=0$ and $\|\bar r\|^2\le K$. One routing layer (Section~\ref{subsec:phase-network}) maps tokens $h_1,\dots,h_P\in\mathbb R^d$ as
\begin{equation*}
\begin{gathered}
u_p=h_p+\pi_p,\quad
\rho'_j=\operatorname{Attn}_s(\rho_j;u_{1:P}),\\
m_p=\operatorname{Attn}_r(u_p;\rho'_{1:M}),
\end{gathered}
\end{equation*}
\begin{equation*}
o_p=\LN_1(u_p+m_p),\qquad h_p^+=\LN_2\big(o_p+\operatorname{MLP}(o_p)\big),
\end{equation*}
with routers $\rho_j$, positional embeddings $\pi_p$, and $\operatorname{Attn}(z;s_{1:n})=W_o\operatorname{concat}_{\text{heads}}\big(\sum_i\alpha_i(W_vs_i+c_v)\big)+c_o$, where the weights $\alpha_i$ are a softmax over $i$ of $\langle W_qz+c_q,W_ks_i+c_k\rangle$ per head. The first layer receives the embedding $h^{(0)}_p=\LN_e(W_e\tilde x_p+b_e)$. In a stack of $N_r$ layers, layer $l+1$ receives $h^{(l)}_p+\LN'_l\big(W'_l(U_lh^{(l)}_p+c_{U,l})+c'_l\big)$, the previous output plus a re-embedding of its projection $U_lh^{(l)}_p+c_{U,l}\in\mathbb R^{\Pin}$, so the last operation of every layer is $\LN_2$.

\subsection{The seasonal random-walk forecaster}

For $b\in\mathbb R^K$ with $\one^\top b=1$ and $\|b\|\le1$, the \emph{level-adaptive seasonal forecaster} is
\begin{equation}
G_b(\tilde X)_p=\one_{K'}\Big[\tfrac1K\one_K^\top\tilde x_p+\big(b-\tfrac1K\one_K\big)^\top\bar r\Big],\qquad p=1,\dots,P .
\label{eq:gb}
\end{equation}
Its level is $\lev(G_b)=b^\top\bar r$, because $\one^\top\bar r=0$.

\begin{lemma}[Bayes forecast]
\label{lem:bayes}
Under model~\equref{eq:mult_model} with $\alpha=0$, Gaussian level increments and noise, and flat priors on the seasonal profile and the initial level, $G_b$ with the Kalman weights of the local level model on the cycle means is the posterior-mean forecast of every future cycle. It is linear and satisfies $G_b(aX+c\one\one^\top)=aG_b(X)+c\one\one^\top$, so forecasting in RevIN coordinates and de-normalizing reproduces it on the raw window. In the steady state the weights are $b_k\approx\kappa(1-\kappa)^{K-k}$ with $\kappa=\tfrac12(-\tau+\sqrt{\tau^2+4\tau})$ and $\tau=P\sigma_\eta^2/\sigma_e^2$.
\end{lemma}

\begin{proof}
The cycle means $r_k=\tfrac1P\sum_pX_{kp}=\ell_k+\bar e_k$ and the phase deviations $X_{kp}-r_k=s_p+(e_{kp}-\bar e_k)$ are independent for Gaussian noise, and the deviations do not involve $\ell$. The deviations identify $s_p$ with posterior mean $\tfrac1K\sum_k(X_{kp}-r_k)$. The cycle means form a local level model with observation variance $\sigma_e^2/P$, whose posterior mean at every future cycle is the Kalman forecast $b^\top r$. Adding the two posterior means gives~\equref{eq:gb} on raw data. Linearity is immediate, and $(b-\tfrac1K\one)^\top\one=0$ gives the shift property. \qed
\end{proof}

\subsection{Proof of Proposition~\ref{prop:capability}}

\fakeparagraph{Full statement} Consider \sysname with one routing layer, latent width $d\ge4$, and any number of routers and heads.
(a) It is exactly equivariant to the level and scale of the window: $F(ax+c\one)=aF(x)+c\one$ for $a>0$.
(b) A cycle-level trajectory enters every phase token as a common shift, and router attention passes a common shift of its inputs on: sender weights are unchanged and every router buffer shifts by the same vector (Lemma~\ref{lem:shift}). The LayerNorms preserve a common shift only approximately, which part~(c) controls with a large offset.
(c) For every weight vector $b$ with $\one^\top b=1$ and $\|b\|\le1$, there are parameters $\theta_c$, $c^2\ge5L$, whose forecast approximates $G_b$ uniformly over all normalized windows, with error at most $\kappa_0L^{3/2}/c^2$ for an absolute constant $\kappa_0$. By Lemma~\ref{lem:bayes}, \sysname can therefore approximate the Bayes forecast of model~\equref{eq:mult_model} with $\alpha=0$ to any accuracy.

\begin{lemma}[Common-shift pass-through]
\label{lem:shift}
If every input token of a routing layer is shifted by the same vector, $u_p\mapsto u_p+v$, then every sender weight is unchanged, every router buffer shifts by $V_sv$ with $V_s=W^s_oW^s_v$, and every receiver output shifts by $V_rV_sv$ plus the effect of the receiver re-weighting the unshifted buffers, with $V_r=W^r_oW^r_v$.
\end{lemma}

\begin{proof}
The sender score of router $j$ and token $p$ changes by $\langle W_q\rho_j+c_q,W_kv\rangle$ in every head, which does not depend on $p$ and cancels in the softmax over $p$. The buffer is a convex combination of $W_v(u_p+v)+c_v$ followed by $W_o$, which gives the buffer shift. The receiver weights may change, but they are convex and every buffer is shifted by the same vector, so the output shifts by $V_rV_sv$, while the re-weighting acts only on the unshifted buffers. \qed
\end{proof}

\begin{lemma}[Large-offset linearization of LayerNorm]
\label{lem:ln_linear}
Let $e_0\perp\one_d$ with $\|e_0\|^2=d$, $w\perp\{\one_d,e_0\}$, $\gamma=\one$, $\beta=0$ and $c>0$. Then $c\LN(ce_0+w)=(1-\eta)(ce_0+w)$ with $0\le\eta\le\rho^2/(2c^2)$ and $\rho^2=\|w\|^2/d+\epsilon$.
\end{lemma}

\begin{proof}
$J_d(ce_0+w)=ce_0+w$ and $\|ce_0+w\|^2/d=c^2+\|w\|^2/d$, so the LayerNorm scale is $c(1+\rho^2/c^2)^{1/2}$, and $1-(1+t)^{-1/2}\le t/2$. \qed
\end{proof}

\begin{proof}[of Proposition~\ref{prop:capability}]
(a) RevIN maps $ax+c\one$ to the same $\tilde x$, and de-normalization restores $a\sigma$ and $a\mu+c$. (b) A cycle-level trajectory $\ell\in\mathbb R^K$ adds $\ell$ to every phase token, hence $W_e\ell$ to every embedding pre-activation; Lemma~\ref{lem:shift} applies to any common shift of the inputs of a routing layer.

(c) Choose orthogonal $e_0,e_1,e_2\perp\one_d$ with $\|e_i\|^2=d$, which is possible because $d\ge4$. Let $\alpha_1=\tfrac1K\one_K$, $\alpha_2=b-\tfrac1K\one_K$, $f_p=(\alpha_1-\alpha_2)^\top\tilde x_p$ and $g_p=\alpha_2^\top\tilde x_p$. Set $W_e=e_1(\alpha_1-\alpha_2)^\top+e_2\alpha_2^\top$, $c_e=ce_0$, all LayerNorm gains one and biases zero, no positional embeddings, $W^s_q=0$ (uniform sender weights), $W^s_oW^s_v=\tfrac1de_2e_2^\top$, $W^r_oW^r_v=I$, all attention biases zero, a zero MLP output layer, and $\Phi=\tfrac{c_*}{d}\one_{K'}(e_1+e_2)^\top$, $c_\Phi=0$, where $c_*=c(1+O(\epsilon))$ makes the map exact at zero content. The embedding gives $h_p=(ce_0+f_pe_1+g_pe_2)/n_p$ with $n_p=(c^2+f_p^2+g_p^2+\epsilon)^{1/2}$. Uniform sender weights make every router buffer equal to $\bar g_he_2$ with $\bar g_h=\tfrac1P\sum_qg_q/n_q$, and since all buffers are equal, every receiver output equals this buffer for any receiver weights. The predictor reads the $e_1+e_2$ coordinate and returns $\one_{K'}(c_*/M_p)(f_p+g_p+n_p\bar g_h)$, where $M_p$ collects the LayerNorm scales. Since $f_p+g_p=\alpha_1^\top\tilde x_p$ and $\tfrac1P\sum_qg_q=\alpha_2^\top\bar r$, the zero-error limit is $G_b(\tilde X)_p$. For the error, $\|\alpha_1-\alpha_2\|^2=\|b\|^2\le1$, $\|\alpha_2\|^2\le1$ and $\|\tilde x_p\|\le\sqrt L$, so $|f_p|,|g_p|,|\bar g_h|\le\sqrt L$ and every scale argument is at most $5L/c^2$. The three LayerNorm factors and the ratios $n_p/n_q$ each contribute a relative error of at most half of this by Lemma~\ref{lem:ln_linear}, and multiplying by the target magnitude $3\sqrt L$ gives the bound with $\kappa_0\le30$. With Lemma~\ref{lem:bayes}, \sysname approximates the Bayes forecast. \qed
\end{proof}

A direct numerical implementation of the construction with $K=30$, $P=24$, $d=8$ and Kalman weights ($\tau=0.05$) on random-walk windows gives $\max|F-G_b|=0.54$, $0.075$, $0.0078$, $7.6\times10^{-4}$ and $8.2\times10^{-5}$ for $c=3,10,30,100,300$, the predicted $c^{-2}$ rate.

\subsection{Proof of Theorem~\ref{thm:level_envelope} and Its Corollaries}

\fakeparagraph{Full statement} For every parameter value of \sysname and every input window:
(a) every normalized phase forecast lies in the fixed ellipsoid $\mathcal E_\theta=\{o_\varphi+A\varsigma:\ \varsigma\perp\one_d,\ \|\varsigma\|<\sqrt d\}$;
(b) the forecast level lies in the envelope, $\lev(\hat{\tilde y})\in\mathcal I_\varphi$, so that $|\mu_{\hat y}-\mu|<(|c_0|+C_1)\sigma$ for the mean $\mu_{\hat y}$ of the forecast in original units;
(c) every window with level demand $\Lambda$ satisfies~\equref{eq:level_floor};
(d) the level and the shape of a phase compete for the same coordinates: as the level of phase $p$ approaches the boundary of $\mathcal I_\varphi$, the input-dependent part of its forecast is forced onto the single profile $AJ_dA^\top\one$.
Parts~(b) and~(c) are Theorem~\ref{thm:level_envelope}.

\begin{proof}
Every routing layer ends with $\LN_2$, so $h^{(N_r)}_p=\gamma_2\odot\varsigma_p+\beta_2$ with $\varsigma_p\perp\one_d$ and $\|\varsigma_p\|^2=d\|J_dz\|^2/(\|J_dz\|^2+d\epsilon)<d$. Substituting into~\equref{eq:predictor} gives $F_p=o_\varphi+A\varsigma_p$, and $\hat{\tilde y}=F$ for \sysname, which is (a). For a functional $a\in\mathbb R^{K'}$, $\langle a,A\varsigma_p\rangle=\langle J_dA^\top a,\varsigma_p\rangle$ is bounded by $\sqrt d\|J_dA^\top a\|$. With $a=\one/K'$, the level $\ell_p=\one^\top F_p/K'$ of phase $p$ lies in $\mathcal I_\varphi$, and the forecast level is the average of the $\ell_p$, which gives (b). For (c), the Cauchy--Schwarz inequality gives $\tfrac1H\|e\|^2\ge\lev(e)^2$ for the normalized error $e=\hat{\tilde y}-\tilde y$, and $\lev(e)^2=(\lev(\hat{\tilde y})-\Lambda)^2\ge\operatorname{dist}(\Lambda,\mathcal I_\varphi)^2$; de-normalization multiplies by $\sigma^2$. For (d), let $v=J_dA^\top\one_{K'}/K'$, so that $C_1=\sqrt d\|v\|$, and decompose $\varsigma_p$ along $v$ and its complement: $F_p=o_\varphi+\tfrac{\ell_p-c_0}{\|v\|^2}Av+A\varsigma_\perp$ with $\|\varsigma_\perp\|^2<d\big(1-((\ell_p-c_0)/C_1)^2\big)$. \qed
\end{proof}

\begin{corollary}[Capability costs envelope width]
\label{cor:capability_cost}
If \sysname approximates $G_b$ uniformly to within $\epsilon_0$ in forecast level, then $C_1\ge\sqrt{K\|b\|^2-1}-\epsilon_0$.
\end{corollary}

\begin{proof}[of Corollary~\ref{cor:capability_cost}]
$\lev(G_b)=b^\top\bar r$ is maximized over $\one^\top\bar r=0$, $\|\bar r\|^2\le K$ at $s_b=\sqrt K\|J_Kb\|=\sqrt{K\|b\|^2-1}$ by a window that is constant within cycles, and its negative attains $-s_b$. A forecast level within $\epsilon_0$ of both values lies in $\mathcal I_\varphi$ by Theorem~\ref{thm:level_envelope}, so $c_0+C_1>s_b-\epsilon_0$ and $c_0-C_1<-s_b+\epsilon_0$, hence $C_1>s_b-\epsilon_0$. \qed
\end{proof}

\begin{corollary}[Steps and trends]
\label{cor:steps}
A persistent step that occurred $fL$ samples before the forecast origin has level demand tending to $\sqrt{(1-f)/f}$ as the step grows relative to the background, and a pure linear trend that continues through the horizon has $\Lambda=\sqrt3(L+H)/\sqrt{L^2-1}$. Any step more recent than $L/(1+(|c_0|+C_1)^2)$ samples therefore demands a level outside $\mathcal I_\varphi$, and at $L=720$ a continued trend demands $\Lambda=1.96$ to $3.46$ for $H=96$ to $720$.
\end{corollary}

\begin{proof}
For a step of size $\Delta$ at $fL$ samples before the origin over zero-mean background variation of variance $\sigma_0^2$, the window mean is $\Delta f$, the window variance is $\Delta^2f(1-f)+\sigma_0^2$, and the future level is $\Delta$, so $\Lambda=\Delta(1-f)/(\Delta^2f(1-f)+\sigma_0^2)^{1/2}\to\sqrt{(1-f)/f}$, which exceeds $|c_0|+C_1$ when $f<1/(1+(|c_0|+C_1)^2)$. For a linear trend, the time index $1,\dots,L$ has mean $(L+1)/2$ and population variance $(L^2-1)/12$, and the future mean index is $L+(H+1)/2$. \qed
\end{proof}

\subsection{Proof of Theorem~\ref{thm:complement}}
\label{app:proof_complement}

\fakeparagraph{Full statement} Let the phase path $F$ have its level in $\mathcal I_\varphi$, and let a branch read any feature of the window.
(a) The least-squares linear predictor of the residual $\tilde y-F$ is a matrix of rank at most $m$ on the profiles $a_1,\dots,a_m$, plus a term driven only by the shape error of the phase path; the innovation noise contributes nothing.
(b) For Gaussian features, $\lambda_1$ affine in the features and a phase path that clips its level to $\mathcal I_\varphi$, the readout onto $a_1$ is the Bayes level readout scaled by $P(\lambda_1\notin\mathcal I_\varphi)$: the optimal branch acts on the level only to the extent that the envelope binds.
(c) Let $m=1$ with $\lambda_1$ linear in the window, let the branch be anchored, $F_{\mathrm b}=\tilde x_L\one_H+W(\tilde x-\tilde x_L\one_L)$, and let the jointly trained phase path carry the shape at the scale of the gates of~\equref{eq:fusion}, $F'=M_cS/(1-g_\ell)+J_cS/(1-g_s)$. Then a rank-one $W=\one_Hu^\top$ attains the Bayes risk of~\equref{eq:profile_model} for any gates. With equal gates and $m$ profiles, $\operatorname{rank}W\le m$.
Part~(a) is Theorem~\ref{thm:complement} of Section~\ref{sec:theory_level}; we refer to the three parts as Theorem~\ref{thm:complement}(a)--(c).

Write the phase forecast as $F=S+\sum_ja_j\varphi_j+r_\varphi$ with $r_\varphi\perp\operatorname{span}\{a_j\}$, where $r_\varphi$ is the phase path's shape error; then $\varphi_1=\lev(F)\in\mathcal I_\varphi$, and the conditional residual is
\begin{equation}
D=\tilde y-F=\textstyle\sum_ja_j\psi_j+\nu-r_\varphi,\qquad\psi_j=\lambda_j-\varphi_j,
\label{eq:dcond}
\end{equation}
with innovation $\nu=\tilde y-\mathbb E[\tilde y\mid\tilde x]$ and $|\psi_1|\ge\operatorname{dist}(\lambda_1,\mathcal I_\varphi)$.

\begin{proof}
(a) Let $\zeta$ be the feature read by the branch and $\Sigma=\mathbb E[\zeta\zeta^\top]\succ0$. Since $\zeta$ is a function of $\tilde x$, $\mathbb E[\nu\zeta^\top]=0$. Hence $W^*=\mathbb E[D\zeta^\top]\Sigma^{-1}=\sum_ja_j\beta_j^\top-\mathbb E[r_\varphi\zeta^\top]\Sigma^{-1}$ with $\beta_j=\Sigma^{-1}\mathbb E[\psi_j\zeta]$, whose first term has rank at most $m$. For any $W$ the excess risk is $H^{-1}\|(W-W^*)\Sigma^{1/2}\|_F^2$; with $W=\sum_ja_j\beta_j^\top$ it equals $H^{-1}\|\mathbb E[r_\varphi\zeta^\top]\Sigma^{-1/2}\|_F^2\le H^{-1}\mathbb E\|r_\varphi\|^2$, and by Eckart--Young the predictive singular values beyond the $m$-th have total energy at most $\|\mathbb E[r_\varphi\zeta^\top]\Sigma^{-1/2}\|_F^2$.

(b) For $\zeta\sim\mathcal N(0,\Sigma)$, Stein's lemma gives $\beta_1=\mathbb E[\nabla_\zeta\psi_1]$. If $\lambda_1=w^\top\zeta+\text{const}$ and the phase path clips its level to $\mathcal I_\varphi$, $\psi_1$ is the soft-threshold excess of $\lambda_1$ over $\mathcal I_\varphi$, whose gradient is $\mathbb 1\{\lambda_1\notin\mathcal I_\varphi\}w$ almost everywhere, so $\beta_1=P(\lambda_1\notin\mathcal I_\varphi)\,w$.

(c) Let $\lambda_1=w^\top\tilde x$; since $\one^\top\tilde x=0$, we may take $\one^\top w=1$. Set $W=\one_Hu^\top$ with $u=(w-(1-g_\ell)L^{-1}\one_L)/g_\ell$, so that $\one^\top u=1$. The anchor cancels, $F_{\mathrm b}=\tilde x_L\one_H+\one_H(u^\top\tilde x-\tilde x_L)=\one_Hu^\top\tilde x$, and $\one^\top\tilde x=0$ gives $g_\ell u^\top\tilde x=w^\top\tilde x$. The branch output is constant over the horizon, so $M_cF_{\mathrm b}=F_{\mathrm b}$ and $J_cF_{\mathrm b}=0$. By~\equref{eq:fusion} with $F'=M_cS/(1-g_\ell)+J_cS/(1-g_s)$, $M_c\hat{\tilde y}=M_cS+\one_H\lambda_1$ and $J_c\hat{\tilde y}=J_cS$, so $\hat{\tilde y}=S+\one_H\lambda_1=\mathbb E[\tilde y\mid\tilde x]$ for any gates. With equal gates $g$ and any other split of the level between the two paths, the branch must supply $[\mathbb E[\tilde y\mid\tilde x]-(1-g)F']/g-\tilde x_L\one$, which lies in $\operatorname{span}\{a_j\}$ when $F'$ carries the shape, so the rank is at most $m$. \qed
\end{proof}

\fakeparagraph{The cycle-mean branch contains the construction}
Under model \equref{eq:mult_model} with $\alpha=0$, Lemma~\ref{lem:bayes} gives $\lambda_1=b^\top\bar r=w^\top\tilde x$ with $w=\tfrac1Pb\otimes\one_P$, which is constant within cycles, and so is $u$ in part~(c). \subsection{Proof of Theorem~\ref{thm:amplitude}}

Index the forecast by future cycle $k'$ and phase $p$. The projection $M_c$ replaces every entry by the mean of its future cycle, and $J_c=I-M_c$. Since $\one_H$ is constant within cycles, $J_c\one_H=0$. Since $U=J_c\operatorname{span}\{a_2,a_3\}\subseteq\operatorname{range}(J_c)$, the operator $\Pi=J_c-P_U$ is the orthogonal projection onto $\operatorname{range}(J_c)\cap U^\perp$, and $\Pi J_c=\Pi$.

\begin{proof}
(a) By part~(a) of the full statement of Theorem~\ref{thm:level_envelope}, the phase forecast of phase $p$ is $F_p=o_\varphi+A\varsigma_p$ with $\varsigma_p\perp\one_d$ and $\|\varsigma_p\|^2<d$. The common term $o_\varphi$ cancels in the within-cycle deviations, so, as a $K'\times P$ array, $J_cF=A\,[\varsigma_1-\bar\varsigma,\dots,\varsigma_P-\bar\varsigma]=AJ_d\,\Sigma_c$ with $\bar\varsigma=\tfrac1P\sum_p\varsigma_p$ and centered columns $\Sigma_c$. Then $\|\Sigma_c\|_F^2=\sum_p\|\varsigma_p\|^2-P\|\bar\varsigma\|^2<Pd$, so $\|J_cF\|^2\le\|AJ_d\|_2^2\|\Sigma_c\|_F^2<Pd\|AJ_d\|_2^2$ and $\tfrac1{\sqrt H}\|J_cF\|<\sqrt{d/K'}\,\|AJ_d\|_2=B_\varphi$, which is the claim for \sysname, whose forecast is $\hat{\tilde y}=F$. The branch output~\equref{eq:branch} is $F_{\mathrm b}=\tilde x_L\one_H+\Psi_H(\Gamma t+c_\Gamma)$; its within-cycle part lies in $J_c\operatorname{span}\{\one_H,a_2,a_3\}=U$, so $\Pi F_{\mathrm b}=0$. By~\equref{eq:fusion}, $\Pi\hat{\tilde y}=\Pi J_c\hat{\tilde y}=(1-g_s)\Pi F+g_s\Pi F_{\mathrm b}=(1-g_s)\Pi F$, and $\|\Pi F\|\le\|J_cF\|$. For the error, $\tfrac1H\|\hat y-y\|^2\ge\tfrac{\sigma^2}{H}\|\Pi(\hat{\tilde y}-\tilde y)\|^2$ because $\Pi$ is a projection that annihilates $\mu\one_H$, and the reverse triangle inequality gives $\|\Pi(\hat{\tilde y}-\tilde y)\|/\sqrt H\ge\Omega-(1-g_s)B_\varphi$.

(b) In raw units the window is $X_{kp}=\ell_k+(1+\alpha\ell_k)s_p$ with $\sum_ps_p=0$, so the cycle means are $\ell_k$ and the window mean is $\mu=\tfrac1K\sum_k\ell_k$. The window's average within-cycle shape in normalized units is $\tfrac1K\sum_k(X_{kp}-\ell_k)/\sigma=(1+\alpha\mu)s_p/\sigma$; denote it, tiled over the horizon, by $\bar s$. With a persistent future level $\ell_f$, the future is $y_{k'p}=\ell_f+(1+\alpha\ell_f)s_p$, so $\Lambda=(\ell_f-\mu)/\sigma$ and
\begin{equation*}
J_c\tilde y=\frac{(1+\alpha\ell_f)\,s}{\sigma}
=\frac{1+\alpha\mu+\alpha\sigma\Lambda}{1+\alpha\mu}\,\bar s
=(1+\alpha_w\Lambda)\,\bar s ,
\end{equation*}
with $\alpha_w=\alpha\sigma/(1+\alpha\mu)$, assuming $1+\alpha\mu>0$. Applying $\Pi=\Pi J_c$ gives $\Omega=|1+\alpha_w\Lambda|\,\|\Pi\bar s\|/\sqrt H$. \qed
\end{proof}



\subsection{Proof of Proposition~\ref{prop:modulation}}

\fakeparagraph{Full statement} Let the final phase latents be modulated as in~\equref{eq:la_modulation}, $h'_p=h_p+(h_p-\bar h)\odot q$, with any $q\in\mathbb R^d$ that may depend on the window.
(a) Every future cycle mean of the phase forecast is unchanged, so Theorems~\ref{thm:level_envelope} and~\ref{thm:complement} hold for the modulated phase path, and the level path of \sysL is untouched.
(b) A vector $q=q_1\one_d$ multiplies the amplitude envelope by $|1+q_1|$. If the phase path carries the window's average shape, $\Pi F=\Pi\bar s/(1-g_s)$, and the branch state reads a level forecast $\hat\Lambda$ linearly, $\xi^\top t=\hat\Lambda$ for some $\xi\in\mathbb R^3$, then $q=\alpha_w\one_d\,\xi^\top t$ yields the within-cycle forecast $(1+\alpha_w\hat\Lambda)\,\Pi\bar s$, the target of Theorem~\ref{thm:amplitude}(b) with $\Lambda$ replaced by its forecast. This $q$ has the form $V_2V_1t$ of~\equref{eq:la_modulation} with $k_q=1$, $V_1=\xi^\top$ and $V_2=\alpha_w\one_d$, so a modulation of width one suffices.

\begin{proof}
(a) $\sum_p(h_p-\bar h)\odot q=\big(\sum_ph_p-P\bar h\big)\odot q=0$. The predictor is affine and shared, so $\tfrac1P\sum_p(\Phi h'_p+c_\Phi)=\tfrac1P\sum_p(\Phi h_p+c_\Phi)$ for every future cycle. The fusion~\equref{eq:fusion} uses the phase forecast only through $M_cF$ in its level part, so the level of the fused forecast is unchanged as well. The level envelope applies because the modulated forecast has the same level as the unmodulated forecast computed from the same latents, which lies in $\mathcal I_\varphi$, and Theorem~\ref{thm:complement} depends on the phase path only through its level and its shape error.

(b) With $h_p=\gamma_2\odot\varsigma_p+\beta_2$, the modulated deviations are $(h_p-\bar h)\odot(\one+q)=\operatorname{diag}(\gamma_2\odot(\one+q))J_d(\varsigma_p-\bar\varsigma)$, and the argument of Theorem~\ref{thm:amplitude}(a) gives the envelope $B_\varphi(q)=\sqrt{d/K'}\,\|\Phi\operatorname{diag}(\gamma_2\odot(\one+q))J_d\|_2$; for $q=q_1\one_d$ the diagonal factor is the scalar $1+q_1$, so $B_\varphi(q)=|1+q_1|B_\varphi$. If $q=\alpha_w\one_d\,\xi^\top t=\alpha_w\hat\Lambda\one_d$, then the modulated phase forecast $F^{q}$ satisfies $J_cF^{q}=(1+\alpha_w\hat\Lambda)J_cF$ and, with $\Pi F=\Pi\bar s/(1-g_s)$, $\Pi\hat{\tilde y}=(1-g_s)\Pi F^{q}=(1+\alpha_w\hat\Lambda)\Pi\bar s$. The modulation~\equref{eq:la_modulation} realizes this $q$ with $w=k_q=1$ (and hence with any $k_q\ge1$), $V_1=\xi^\top$ and $V_2=\alpha_w\one_d$. \qed
\end{proof}

The modulation thus removes the constant amplitude envelope along exactly the required direction without touching the level.
